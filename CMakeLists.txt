cmake_minimum_required(VERSION 3.16)
project(RecompVR LANGUAGES C)

add_library(recompvr STATIC src/openxr.c)
add_library(RecompVR::openxr ALIAS recompvr)
target_compile_features(recompvr PUBLIC c_std_99)
target_include_directories(recompvr PUBLIC include)
if(TARGET OpenXR::headers)
    target_link_libraries(recompvr PUBLIC OpenXR::headers)
else()
    if(NOT DEFINED RECOMPVR_OPENXR_INCLUDE_DIR)
        set(RECOMPVR_OPENXR_INCLUDE_DIR "" CACHE PATH "Directory containing openxr/openxr.h")
    endif()
    if(NOT EXISTS "${RECOMPVR_OPENXR_INCLUDE_DIR}/openxr/openxr.h")
        message(FATAL_ERROR "Set RECOMPVR_OPENXR_INCLUDE_DIR to the OpenXR headers")
    endif()
    target_include_directories(recompvr PUBLIC "${RECOMPVR_OPENXR_INCLUDE_DIR}")
endif()
set_target_properties(recompvr PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(RECOMPVR_WITH_VULKAN "Build XR_KHR_vulkan_enable2 helpers" OFF)
if(RECOMPVR_WITH_VULKAN)
    target_sources(recompvr PRIVATE src/vulkan.c)
endif()
if(ANDROID)
    target_sources(recompvr PRIVATE src/android.c)
endif()

option(RECOMPVR_BUILD_TESTS "Build the host OpenXR contract test" OFF)
if(RECOMPVR_BUILD_TESTS)
    enable_testing()
    add_executable(recompvr_openxr_test tests/openxr_test.c)
    target_compile_options(recompvr_openxr_test PRIVATE -UNDEBUG)
    target_link_libraries(recompvr_openxr_test PRIVATE RecompVR::openxr)
    add_test(NAME recompvr_openxr_test COMMAND recompvr_openxr_test)
endif()
